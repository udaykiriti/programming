// lgammabf16.sollya
// Script to generate LGAMMA_POLY coefficients for lgammabf16.h
// Fits a degree-4 polynomial to lgamma(x) on each interval [n, n+1)
// centered at n+0.5, using fpminimax for near-minimax float32 coefficients.
//
// Note: interval [1,2) is special-cased with absolute error, since
// lgamma(x) has exact zeros at both x=1 and x=2, which breaks
// relative-error fitting (division by zero at both endpoints).
//
// Run with: sollya lgammabf16.sollya

f := log(gamma(x));

for n from 1 to 7 do {
    center := n + 0.5;
    g := f(x + center);

    if (n == 1) then {
        P := fpminimax(g, [|0,1,2,3,4|], [|24,24,24,24,24|], [-0.5, 0.5], absolute);
        err := supnorm(P, g, [-0.5, 0.5], absolute, 2^(-50));
    } else {
        P := fpminimax(g, [|0,1,2,3,4|], [|24,24,24,24,24|], [-0.5, 0.5], relative);
        err := supnorm(P, g, [-0.5, 0.5], relative, 2^(-50));
    };

    print("// [", n, ",", n+1, "), center=", center, ", max_rel_err=", err);
    for i from 0 to 4 do {
        printsingle(coeff(P, i));
    };
    print("");
};

// evaluated as P(t) = c0 + c1*t + c2*t^2 + c3*t^3 + c4*t^4

procedure evalpoly(c0, c1, c2, c3, c4, t) {
    return c0 + t*(c1 + t*(c2 + t*(c3 + t*c4)));
};
